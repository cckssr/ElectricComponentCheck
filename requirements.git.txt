# Beispiel requirements.txt für die Installation von ElectricComponentCheck
# aus einem Git-Repository

# Von GitHub (öffentliches Repository)
# git+https://github.com/yourusername/ElectricComponentCheck.git

# Von GitHub (privates Repository mit SSH)
# git+ssh://git@github.com/yourusername/ElectricComponentCheck.git

# Von einem spezifischen Branch
# git+https://github.com/yourusername/ElectricComponentCheck.git@develop

# Von einem spezifischen Tag/Version
# git+https://github.com/yourusername/ElectricComponentCheck.git@v0.1.0

# Von einem spezifischen Commit
# git+https://github.com/yourusername/ElectricComponentCheck.git@abc123def456

# Von GitLab
# git+https://gitlab.com/yourusername/ElectricComponentCheck.git

# Von einem selbst gehosteten Git-Server
# git+https://git.your-domain.com/yourusername/ElectricComponentCheck.git

# Mit Authentifizierung (Token)
# git+https://<token>@github.com/yourusername/ElectricComponentCheck.git

# Für editable/development Installation
# -e git+https://github.com/yourusername/ElectricComponentCheck.git#egg=electric-component-check
